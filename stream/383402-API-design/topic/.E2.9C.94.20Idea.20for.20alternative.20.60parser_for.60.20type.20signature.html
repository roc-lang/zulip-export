how do you import a where alias from another module ?

# Foo.roc

Foo := [] {
    a.Stringable : where [a.to_str : a -> Str]
}

import Foo exposing [Stringable]

stringify : v -> Str where [v.Stringable]
stringify = |value| value.to_str()

Anton (Aug 07 2026 at 16:24):

Here is a working example @Romain Lepert

Ifc.roc:

Ifc :: [].{
        a.Stringable : where [a.to_str : a -> Str]
        a.Encodable(fmt) : where [a.encode : a, fmt -> List(U8)]
}

Use.roc:

import Ifc exposing [Stringable, Encodable]

Use :: [].{
        # imported alias, unqualified
        stringify : a -> Str where [a.Stringable]
        stringify = |value| value.to_str()

        # imported alias, module-qualified — also fine
        shout : a -> Str where [a.Ifc.Stringable]
        shout = |value| value.to_str()

        # imported alias composed into a locally declared one
        a.Loud : where [a.Stringable, a.shout : a -> Str]

        # parameterized imported alias
        enc : a, fmt -> List(U8) where [a.Encodable(fmt)]
        enc = |value, f| value.encode(f)
}
