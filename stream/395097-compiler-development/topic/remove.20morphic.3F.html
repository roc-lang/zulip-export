prime_bench = |start| {
  loopi = |cmps, i| {
      s0 = (i + i) * (i + 3) + 3
      if (s0 >= 131072) cmps else {
          elem = cmps.get(i.shr_zf_wrap(3)).ok_or(0.U8)
          msk = 1.U8.shl_wrap(i.bitwise_and(7).to_u8_wrap())
          if (elem.bitwise_and(msk) > 0) loopi(cmps, i + 1) else {
              bp = i + i + 3
              slmt = s0 + bp.shl_wrap(3)
              loops = |s, cs| {
                  if (s >= slmt) cs else {
                      smsk = 1.U8.shl_wrap(s.bitwise_and(7).to_u8_wrap())

# Build with Roc nightly-2026-11-13-03dce41 or later:
#   roc build examples/prime-bench.roc
# Run: ./examples/prime-bench [seed=0] [repeats=100]
app [main!] {
    cli: platform "https://github.com/roc-lang/basic-cli/releases/download/0.28.1/7brgwR8YUqfeGrkdfF1V4GPPN7DnP7ftqub9KcrTEZtr.tar.zst",
}

import cli.Stdout
import cli.Monotonic
import cli.Duration

prime_bench : U64, U64 -> I64
prime_bench = |seed, repeats| {
    # Bit i represents the odd number 2*i + 3; prime 2 is counted separately.
    var $composites = List.repeat(0.U8, 16384)
    var $repeat = 0.U64
    while $repeat < repeats {
        var $i = seed
        # i = 254 is the last index whose prime square fits in the bitmap.
        while $i < 255 {
            elem = match $composites.get($i.shr_zf_wrap(3)) {
                Ok(byte) => byte
                Err(_) => crash "Sieve candidate index out of bounds"
            }
            mask = (1.U8).shl_wrap($i.bitwise_and(7).to_u8_wrap())
            if elem.bitwise_and(mask) == 0 {
                prime = $i + $i + 3
                square_index = ($i + $i) * ($i + 3) + 3
                limit = square_index + prime.shl_wrap(3)
                var $s = square_index
                # Eight streams: each visits one fixed bit in successive bytes.
                while $s < limit {
                    stream_mask = (1.U8).shl_wrap($s.bitwise_and(7).to_u8_wrap())
                    var $byte_index = $s.shr_zf_wrap(3)
                    while $byte_index < 16384 {
                        old_byte = match $composites.get($byte_index) {
                            Ok(value) => value
                            Err(_) => crash "Sieve stream index out of bounds"
                        }
                        # Consume the old list; retaining it as an error default
                        # would require preserving its contents across the update.
                        $composites = match $composites.set($byte_index, old_byte.bitwise_or(stream_mask)) {
                            Ok(updated) => updated
                            Err(_) => crash "Sieve stream index out of bounds"
                        }
                        $byte_index = $byte_index + prime
                    }
                    $s = $s + prime
                }
            }
            $i = $i + 1
        }
        # Preserve the original benchmark: repeat culling the same bitmap.
        $repeat = $repeat + 1
    }
    var $count = 131073.I64
    var $index = 0.U64
    while $index < 16384 {
        byte = match $composites.get($index) {
            Ok(value) => value
            Err(_) => crash "Sieve count index out of bounds"
        }
        $count = $count - byte.count_one_bits().to_i64()
        $index = $index + 1
    }
    $count
}

    # Nonzero seeds skip sieve candidates; use zero to count all primes.
    seed = match args.get(0) {
        Ok(arg) => U64.from_str(arg.display()).ok_or(0)
        Err(_) => 0
    }

    }

}
