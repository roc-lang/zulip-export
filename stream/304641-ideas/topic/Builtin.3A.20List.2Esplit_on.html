split_on : List(item), (item, item -> Bool) -> List(List(item))
split_on = |items, fun| {
    aux = |rest, inner_acc, acc| match rest {
        [] => acc
        [h] => acc.append(inner_acc.append(h))
        [f, s, .. as t] => {
            if (fun(f, s)) {
                aux(t.prepend(s), inner_acc.append(f), acc)
            } else {
                aux(t.prepend(s), [], acc.append(inner_acc.append(f)))
            }
        }
    }
    aux(items, [], [])
}
expect [1, 2] |> split_on(|cur, next| cur.is_eq(next)) |> List.is_eq([[1], [2]])
expect [1, 1, 2] |> split_on(|cur, next| cur.is_eq(next)) |> List.is_eq([[1, 1], [2]])
expect [1, 1, 2, 3] |> split_on(|cur, next| cur.is_eq(next)) |> List.is_eq([[1, 1], [2], [3]])
