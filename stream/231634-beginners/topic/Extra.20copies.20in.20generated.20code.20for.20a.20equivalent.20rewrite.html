advance : Machine, U64 -> Try(Machine, [Backwards, OutOfBounds])
advance = |m, target| {
    if target < m.now { Err(Backwards) }
    else if target - m.now > 2 {
        changed = m.bus.bytes.set(0, m.bus.bytes.get(0)?.plus_wrap(1))?
        advance({ ..m, now: m.now + 2, bus: { ..m.bus, bytes: changed } }, target)
    } else { Ok({ ..m, now: target }) }
}

read16 : Bus, U32 -> Try(ReadResult(U16), [OutOfBounds, ReadError])
read16 = |bus, addr| {
    # Keep a non-inline read shape. The exercised addresses are 0..20.
    if addr >= 32 {
        changed = bus.bytes.set(0, bus.bytes.get(0)?.plus_wrap(1))?
        return read16({ ..bus, bytes: changed }, addr - 32)
    }
    val = bus.bytes.get(addr.to_u64())?
    match addr.mod_by(4) {
        0 => Ok({ val: val.to_u16(), state: Unchanged })
        1 => Ok({ val: val.to_u16(), state: Updated(Box.box({ ..bus, bytes: bus.bytes.set(addr.to_u64(), val.plus_wrap(1))? })) })
        2 => Ok({ val: val.to_u16(), state: Unchanged })
        _ => Err(ReadError)
    }
}
read32 : Bus, U32 -> Try(ReadResult(U32), [OutOfBounds, ReadError])
read32 = |bus, addr| {
    first = read16(bus, addr)?
    Ok({ val: first.val.to_u32(), state: first.state })
}
