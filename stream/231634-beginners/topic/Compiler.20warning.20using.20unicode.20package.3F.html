I'm just posting the whole function for convenience:

next_range : RangeIterState -> Try((ByteRange, RangeIterState), [NoMore])
next_range = |state| {
    if state.finished {
        return Err(NoMore)
    }

    var $cursor = state.cursor
    var $machine = state.machine

    while Bool.True {
        match InternalUtf8.next($cursor) {
            Done => {
                if $machine.started {
                    range = ByteRange.from_bounds($machine.cluster_start, $cursor.byte_offset) ?? ...
                    return Ok((
                        range,
                        {
                            cursor: $cursor,
                            machine: $machine,
                            finished: Bool.True,
                        },
                    ))
                } else {
                    return Err(NoMore)
                }
            }
            One({ item, rest }) => {
                transition = InternalGrapheme.push($machine, item.scalar, item.byte_start)
                $cursor = rest
                $machine = transition.machine

                match transition.boundary {
                    NoBoundary => {}
                    Boundary({ start, end }) => {
                        range = ByteRange.from_bounds(start, end) ?? ...
                        return Ok((
                            range,
                            {
                                cursor: $cursor,
                                machine: $machine,
                                finished: Bool.False,
                            },
                        ))
                    }
                }
            }
        }
    }
}
