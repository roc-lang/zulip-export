Adityavarman Manjunath (Sep 28 2026 at 18:56):

just got a good base setup for helix if someone is looking for it:

use-grammars = { only = ["roc"] }

[language-server.roc-lsp]
command = "roc"
args = ["experimental-lsp", "--stdio"]

[[language]]
name = "roc"
scope = "source.roc"
injection-regex = "roc"
file-types = ["roc"]
shebangs = ["roc"]
roots = []
comment-token = "#"
language-servers = ["roc-lsp"]
indent = { tab-width = 4, unit = "    " }
formatter = { command = "roc", args = ["fmt", "--stdin"] }
auto-format = true

[language.auto-pairs]
'(' = ')'
'{' = '}'
'[' = ']'
'"' = '"'

[[grammar]]
name = "roc"
source = { git = "https://github.com/faldor20/tree-sitter-roc.git", rev = "master"}
