import pf.Path
import pf.Stdout
import pf.Stderr
import pf.Url
import pf.OsStr
import pf.Http

import http.Method
import http.Request

main! : List(OsStr) => Try({}, [Exit(I32), ..])
main! = |args| {
    match run!(args) {
        Ok({}) => Ok({})
        Err(err) => {
            _ = Stderr.line!("Runtime error: ${Str.inspect(err)}")

            Err(Exit(2))
        }
    }
}

run! = |_args| {
    uri = "https://"

    _ = parse_uri(uri)?
    request = Request.from_method(GET).with_uri(uri)

    Stdout.line!("${request.method().to_str()} ${request.uri()}")?

    response = Http.send!(request)?

    response.headers().for_each!(
        |header| {
            dbg "${header.name}: ${header.value}"
        },
    )

    dbg response.status().to_str()
    dbg Str.from_utf8(response.body())?

    Ok({})
}

parse_uri : Str -> Try(Url, [InvalidUrl(Url.ParseErr), ..])
parse_uri = |uri| {
    match Url.parse(uri) {
        Ok(url) => Ok(url)
        Err(err) => Err(InvalidUrl(err))
    }
}

Here's another example:

Ideally, the inferred type would be [BadUtf8, RequestBodyErr] or something like that, but I get a type mismatch. I suppose, like @boof , it's because read_all!'s error union is closed?

Aurélien Geron (Aug 23 2026 at 21:03):

Yes, it's probably because read_all! returns a closed union. You can work around this issue by using (x.read_all!() ? |RequestBodyErr(err)|RequestBodyErr(err)). Not ideal, but hopefully this will become automatic in the future.
